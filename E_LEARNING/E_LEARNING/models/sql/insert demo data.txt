-- 2) Générer 3 chapitres par cours (titres standards : Fondamentaux, Méthodes, Applications)
INSERT INTO course_chapter (course_id, chapter_title, num_of_reading, num_of_contents, is_visible, sequence)
SELECT c.id,
       CASE n
         WHEN 1 THEN 'Fondamentaux'
         WHEN 2 THEN 'Méthodes'
         WHEN 3 THEN 'Applications pratiques'
       END,
       0,
       2,
       b'1',
       n
FROM course c
CROSS JOIN (SELECT 1 AS n UNION SELECT 2 UNION SELECT 3) chapters;

-- 3) Générer 2 contenus par chapitre avec texte réel généré à la volée (content_type_id = 1 = vidéo)
-- Le texte est construit pour être pédagogique et utilisable tel quel.
INSERT INTO course_chapter_content
(course_chapter_id, content_type_id, is_important, time_required_in_sec, is_open_for_free, is_visible, content_text, content_brief, date_creation, date_update, sequence, content_cover_url, file_size_in_bytes)
SELECT cc.id,
       4,                      -- content_type_id = 1 (texte)
       b'1',
       CASE s.seq WHEN 1 THEN 900 WHEN 2 THEN 600 END, -- durées (sec)
       'Y',
       b'1',
       CONCAT(
         'Cours: ', c.course_title, '\n\n',
         'Chapitre: ', 
           CASE cc.sequence WHEN 1 THEN 'Fondamentaux' WHEN 2 THEN 'Méthodes' ELSE 'Applications pratiques' END,
         '\n\n',
         -- contenu pédagogique : paragraphe unique structuré
         CASE cc.sequence
           WHEN 1 THEN
             'Ce chapitre présente les notions fondamentales. Nous définissons les concepts clés, le vocabulaire et le contexte technique. Vous apprendrez la terminologie, les métriques d''évaluation et verrez des exemples concrets pour comprendre pourquoi ces méthodes sont utiles.'
           WHEN 2 THEN
             'Ce chapitre décrit les méthodes principales et leurs fondements mathématiques. Nous couvrons les algorithmes, le prétraitement des données, le choix des features et les bonnes pratiques d''implémentation. Un petit guide pas-à-pas montre comment implémenter un modèle de base.'
           ELSE
             'Ce chapitre montre des cas pratiques et un mini-projet. Il détaille un pipeline complet, propose des exercices guidés et des pistes d''évaluation pour mesurer la performance en condition réelle.'
         END,
         '\n\n',
         'Exercice / Objectifs : ', 
         CASE s.seq WHEN 1 THEN 'Lecture et compréhension. Résumer les points clés.' WHEN 2 THEN 'Implémentation guidée.' END
       ),
       CONCAT('Contenu: ', 
         CASE cc.sequence WHEN 1 THEN 'Texte introductif' WHEN 2 THEN 'Méthode détaillée' ELSE 'Projet & exercices' END),
       NOW(),
       NOW(),
       s.seq,
       NULL,
       NULL
FROM course_chapter cc
JOIN course c ON cc.course_id = c.id
CROSS JOIN (SELECT 1 AS seq UNION SELECT 2) s;
